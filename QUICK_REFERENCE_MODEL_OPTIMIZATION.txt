│ KEY CONCEPTS                                                                 │
└──────────────────────────────────────────────────────────────────────────────┘

  Thompson Sampling:
    • Bayesian multi-armed bandit algorithm
    • Balances exploration (trying new models) vs exploitation (using best)
    • Automatically learns which models work best per task type

  Quality Scoring (0-1):
    • Validation: 40% weight
    • Static analysis: 20% weight
    • Requirements coverage: 30% weight
    • Human feedback: 10% weight

  Expected Cost Savings:
    • Simple tasks: 90% reduction (Haiku 100%)
    • Medium tasks: 50-70% reduction (Haiku 70%)
    • Complex tasks: 20-30% reduction (Sonnet primary)
    • Overall: 40-60% reduction

│ INTEGRATION EXAMPLE                                                          │
└──────────────────────────────────────────────────────────────────────────────┘

  import { ThompsonSampler } from './ml/thompson-sampler'

  async function executeTask(task: ActivityTask) {
    const sampler = await ThompsonSampler.create()
    
    // Select model
    const selection = await sampler.select(task.type, {
      explorationRate: 0.15,
      candidateModels: [
        { modelID: "claude-sonnet-4-5-20250929", providerID: "anthropic" },
        { modelID: "claude-3-5-haiku-20241022", providerID: "anthropic" }
      ]
    })
    
    // Execute with selected model
    const result = await executeWithModel(task, selection)
    
    // Update statistics
    await sampler.update(
      selection.modelID,
      selection.providerID,
      task.type,
      {
        success: result.validationPassed,
        cost: result.cost,
        qualityScore: computeQualityScore(result)
      }
    )
  }
